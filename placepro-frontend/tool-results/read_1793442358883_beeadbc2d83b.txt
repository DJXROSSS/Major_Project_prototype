    23→/* ------------------------------ seeded RNG ------------------------------ */

    25→function mulberry32(seed: number) {
    26→  let a = seed >>> 0;
    27→  return () => {
    28→    a |= 0;
    29→    a = (a + 0x6d2b79f5) | 0;
    30→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    31→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    32→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    33→  };
    34→}

   211→function domainScore(s: Student, key: "dsa" | "core" | "dev" | "comm"): number {
   212→  const groups: Record<string, string[]> = {
   213→    dsa: ["dsa"],
   214→    core: ["dbms", "os", "cn", "oops"],
   215→    dev: ["java", "springboot"],
   216→    comm: ["comm"],
   217→  };
   218→  const ids = groups[key].filter((id) => s.scores[id] !== undefined);
   219→  return ids.reduce((a, id) => a + s.scores[id], 0) / ids.length;
   220→}

   222→function buildAssessments(): Assessment[] {
   223→  return ASSESSMENT_SEEDS.map((seed, ai) => {
   224→    const batchStudents = STUDENTS.filter((s) => s.batch === seed.batch);
   225→    const weights = FOCUS_WEIGHT[seed.focus];
   226→    const results = batchStudents.map((s) => {
   227→      let expected =
   228→        (Object.keys(weights) as ("dsa" | "core" | "dev" | "comm")[]).reduce(
   229→          (acc, key) => acc + domainScore(s, key) * (weights[key] ?? 0),
   230→          0,
   231→        ) / Object.values(weights).reduce((a, b) => a + (b ?? 0), 0);
   232→      // gentle upward drift for later assessments + noise
   233→      expected = clamp(expected + ai * 0.6 + (rand() - 0.5) * 16, 20, 100);
   234→      const marks = round((expected / 100) * seed.maxMarks, 1);
   235→      return {
   236→        studentId: s.id,
   237→        marks,
   238→        maxMarks: seed.maxMarks,
   239→        percentage: round((marks / seed.maxMarks) * 100, 1),
   240→      };
   241→    });
   242→    return {
   243→      id: `a${ai + 1}`,
   244→      name: seed.name,
   245→      type: seed.type,
   246→      batch: seed.batch,
   247→      date: seed.date,
   248→      uploadedBy: ai % 3 === 0 ? "Priya Nair" : ai % 3 === 1 ? "Dr. Rajesh Menon" : "Vikram Shetty",
   249→      results,
   250→      reviewed: true,
   251→    };
   252→  });
   253→}

   521→export function domainAverage(students: Student[], domain: string): number {
   522→  const group = DOMAIN_GROUPS.find((g) => g.label === domain);
   523→  if (!group) return 0;
   524→  const vals = students.flatMap((s) => group.skills.filter((id) => s.scores[id] !== undefined).map((id) => s.scores[id]));
   525→  return vals.length ? round(vals.reduce((a, b) => a + b, 0) / vals.length, 1) : 0;
   526→}

   528→/** Month-wise batch performance from assessment results */
   529→export function monthlyPerformance(batch: string): { month: string; average: number }[] {
   530→  const assessments = ASSESSMENTS.filter((a) => a.batch === batch);
   531→  const byMonth = new Map<string, number[]>();
   532→  for (const a of assessments) {
   533→    const d = new Date(a.date);
   534→    const label = d.toLocaleDateString("en-IN", { month: "short" });
   535→    const avg = a.results.reduce((s, r) => s + r.percentage, 0) / a.results.length;
   536→    if (!byMonth.has(label)) byMonth.set(label, []);
   537→    byMonth.get(label)!.push(avg);
   538→  }
   539→  return Array.from(byMonth.entries()).map(([month, arr]) => ({
   540→    month,
   541→    average: round(arr.reduce((a, b) => a + b, 0) / arr.length, 1),
   542→  }));
   543→}

   545→export function studentDomainScore(student: Student, domain: string): number {
   546→  const group = DOMAIN_GROUPS.find((g) => g.label === domain);
   547→  if (!group) return 0;
   548→  const ids = group.skills.filter((id) => student.scores[id] !== undefined);
   549→  return ids.length ? round(ids.reduce((a, id) => a + student.scores[id], 0) / ids.length, 1) : 0;
   550→}

   552→export function weakestDomain(student: Student): { domain: string; score: number } {
   553→  const entries = DOMAIN_GROUPS.map((g) => ({ domain: g.label, score: studentDomainScore(student, g.label) }));
   554→  return entries.reduce((min, e) => (e.score < min.score ? e : min), entries[0]);
   555→}
